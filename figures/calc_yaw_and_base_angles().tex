\noindent
$\mathbf v \perp \mathbf l\mathbf t$ at $\mathbf t$. $\mathbf b Q \perp \mathbf y Q$ at $Q$ --
the tangency condition, Eq.~\eqref{eq:tangent}. Teal circle: radius $L_2$ about $\mathbf y$.

\section{Setup}

\begin{align*}
\mathbf b &= \texttt{base\_position}, \quad
\mathbf l = \texttt{lens.position}, \quad
\mathbf t = \texttt{target\_pos} &&\in \mathbb R^3\\
L_2 &= \texttt{L2}
\end{align*}

Only the $xy$-projections matter below; write $\mathbf b,\mathbf l,\mathbf t,\mathbf y$ for them
from here on.

\section{Yaw Axis Position}

\begin{align}
\mathbf d &= \mathbf t - \mathbf l, \qquad
\hat{\mathbf n} = \frac{(-d_y,\ d_x)}{\lVert \mathbf d \rVert} \\
\mathbf v &= L_2\, \hat{\mathbf n}, \qquad
\mathbf y = \mathbf t - \mathbf v
\end{align}

\section{Tangent Condition}

\begin{equation}
\mathbf v_r(\beta) = R(\beta)\,\mathbf v, \qquad
Q(\beta) = \mathbf y + \mathbf v_r(\beta), \qquad
R(\beta) = \begin{bmatrix}\cos\beta & -\sin\beta \\ \sin\beta & \cos\beta\end{bmatrix}
\end{equation}

\begin{equation}
\label{eq:tangent}
\raisebox{-0.5\height}{\tikz \node[draw=teal!70!black, line width=1.4pt, rounded corners=3pt,
fill=teal!8, inner sep=14pt]
{\Large $\displaystyle \mathbf v_r(\beta)\cdot\big(Q(\beta) - \mathbf b\big) = 0$};}
\end{equation}
radius $\mathbf y Q(\beta) \perp$ tangent $\mathbf b Q(\beta)$: $\mathbf v_r(\beta)$, the rotated
offset, is perpendicular to the line from the base to the point it lands on.

\section{Solving \eqref{eq:tangent}}

Write $\mathbf w = \mathbf y - \mathbf b$. Since $\lVert\mathbf v_r(\beta)\rVert = \lVert\mathbf v\rVert = L_2$
for every $\beta$:
\begin{align}
\mathbf v_r(\beta)\cdot\big(Q(\beta)-\mathbf b\big)
&= \mathbf v_r(\beta)\cdot\big(\mathbf w + \mathbf v_r(\beta)\big)
 = \mathbf v_r(\beta)\cdot\mathbf w + L_2^2 \; = \; 0 \\[2pt]
\Longrightarrow\quad \mathbf v_r(\beta)\cdot\mathbf w &= -L_2^2
\end{align}
Substitute $\mathbf v_r(\beta) = (v_x\cos\beta - v_y\sin\beta,\ v_x\sin\beta + v_y\cos\beta)$ and
group by $\cos\beta,\sin\beta$:
\begin{align}
\mathbf v_r(\beta)\cdot\mathbf w
&= (v_x\cos\beta - v_y\sin\beta)\,w_x + (v_x\sin\beta + v_y\cos\beta)\,w_y \\
&= \underbrace{(v_xw_x + v_yw_y)}_{A}\cos\beta \;+\; \underbrace{(v_xw_y - v_yw_x)}_{B}\sin\beta
\end{align}
i.e.\ $A\cos\beta + B\sin\beta = C$ with
\begin{equation}
A = v_x w_x + v_y w_y, \qquad
B = v_x w_y - v_y w_x, \qquad
C = -L_2^2.
\end{equation}

\section{Closed Form}

Divide $A\cos\beta + B\sin\beta = C$ through by $R := \sqrt{A^2+B^2}$:
\begin{equation}
\frac{A}{R}\cos\beta + \frac{B}{R}\sin\beta = \frac{C}{R}
\end{equation}
$(A/R,\,B/R)$ is a unit vector, so it is $(\cos\varphi,\sin\varphi)$ for some $\varphi$ --
read off directly as $\varphi = \operatorname{atan2}(B,A)$:
\begin{equation}
\cos\varphi = \frac{A}{R}, \qquad \sin\varphi = \frac{B}{R}
\end{equation}
Substituting turns the left side into the cosine angle-difference identity:
\begin{equation}
\cos\varphi\cos\beta + \sin\varphi\sin\beta = \cos(\beta - \varphi) = \frac{C}{R}
\end{equation}
Real solutions exist iff $|C| \le R$ (else $\mathbf b$ is unreachable by a tangent from this
circle). Isolating $\beta$:
\begin{equation}
\beta - \varphi = \pm\arccos\!\left(\frac{C}{R}\right)
\end{equation}
\begin{equation}
\boxed{\ \beta = \varphi \pm \arccos\!\left(\frac{C}{R}\right)\ }
\end{equation}

\section{Base Angle}

\begin{equation}
Q = \mathbf y + \mathbf v_r(\beta), \qquad
\theta = \operatorname{atan2}(Q_y - b_y,\ Q_x - b_x)
\end{equation}

The two signs give two $(\beta,\theta)$ pairs; duplicates are dropped and the pair closest to
world heading $0^\circ$ is returned first.
